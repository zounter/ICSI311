class Student
	string firstName, lastName
	number grade1, grade2, grade3
	construct(string first, string last, number homework, number midterm, number final)
		firstName = first
		lastName = last
		grade1 = homework
		grade2 = midterm
		grade3 = final
	shared getAverage(number homework, number midterm, number final) : number average
		average = (homework + midterm + final) / 3
	print()
		console.print("First: " + firstName)
		console.print("Last: " + lastName)
		console.print("Grade:")
		console.print(Student.getAverage(grade1, grade2, grade3))
	shared start()
		Student student1, student2, student3
		student1 = new Student ("Joe", "Doe", 0, 0, 2)
		student2 = new Student ("Jane", "Doe", 87, 65, 97)
		student3 = new Student ("Jill", "Doe", 100, 100, 100)
		student1.print()
		student2.print()
		student3.print()
